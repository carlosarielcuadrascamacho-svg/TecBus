=============================================================================
  CODIGO PARA ESP32 (30 PINES) — TECBUS / SMARTBUS
  Envia ubicacion GPS al servidor de TecBus cada 10 segundos
=============================================================================
  CONEXIONES DEL GPS (NEO-6M / NEO-7M / NEO-8M):
  
    MODULO GPS         ->   ESP32 (30 pines)
    VCC                ->   3.3V (o 5V si tu modulo tiene regulador)
    GND                ->   GND
    TX                 ->   GPIO16 (RX2 de ESP32)
    RX                 ->   GPIO17 (TX2 de ESP32)
  
  NOTA: Muchos modulos NEO-6M funcionan a 3.3V, pero algunos requieren
        5V (tienen regulador integrado). Revisa tu modulo antes de conectar.
  
  LED indicador: Usa el LED integrado (GPIO2) que parpadea segun el estado.
=============================================================================
  CONFIGURACION REQUERIDA EN EL SERVIDOR TECBUS:
  
  El ESP32 usa la ruta HTTP: PUT /api/camiones/update-location
  Envia: { "busId": "U001", "lat": 25.567, "lng": -108.473, "speed": 35 }
  
  Donde "busId" debe ser el NUMERO DE UNIDAD del camion registrado en la BD
  (ej: "U001", "TEC-01", "EC-01", etc.)
=============================================================================

#include <WiFi.h>
#include <HTTPClient.h>
#include <TinyGPSPlus.h>
#include <ArduinoJson.h>

// =========================================================================
//  CONFIGURACION: CAMBIA ESTOS VALORES SEGUN TU CAMION Y RED
// =========================================================================

// --- WiFi (Usa el hotspot del celular del conductor o WiFi del camion) ---
const char* WIFI_SSID = "TecBus_Hotspot";
const char* WIFI_PASS = "conductor123";

// --- Servidor TecBus ---
const char* SERVER_URL = "https://tecbus-api.onrender.com";
// Para pruebas locales:
// const char* SERVER_URL = "http://192.168.1.100:5000";

// --- Identificador del camion (DEBE coincidir con el "numeroUnidad" en MongoDB) ---
const char* BUS_ID = "U001";

// --- Intervalo de envio de ubicacion (milisegundos) ---
const unsigned long INTERVALO_ENVIO_MS = 10000; // 10 segundos

// =========================================================================
//  PINES DEL ESP32 (30 pines - DevKit V1 / DOIT)
// =========================================================================

// El modulo GPS se conecta a UART2 del ESP32
// UART2: RX2 = GPIO16, TX2 = GPIO17
static const int RX2_PIN = 16;
static const int TX2_PIN = 17;
static const int LED_BUILTIN = 2; // LED integrado del ESP32

// =========================================================================
//  OBJETOS GLOBALES
// =========================================================================

TinyGPSPlus gps;
HardwareSerial gpsSerial(2); // UART2

// Estados del LED
enum EstadoLED {
  ESPERANDO_WIFI,    // Parpadeo rapido
  ESPERANDO_GPS,     // Parpadeo lento
  ENVIANDO_DATOS,    // Encendido fijo
  ERROR_CONEXION     // 2 parpadeos, pausa
};

EstadoLED estadoActual = ESPERANDO_WIFI;

// Variables de control de tiempo
unsigned long ultimoEnvio = 0;
unsigned long ultimoParpadeo = 0;
bool ledEstado = false;

// =========================================================================
//  SETUP
// =========================================================================

void setup() {
  Serial.begin(115200);
  Serial.println(F("\n=========================================="));
  Serial.println(F("  TECBUS - ESP32 GPS TRACKER"));
  Serial.println(F("=========================================="));

  // Configurar LED
  pinMode(LED_BUILTIN, OUTPUT);
  digitalWrite(LED_BUILTIN, LOW);

  // Inicializar GPS en UART2 (9600 baudios es lo estandar para NEO-6M)
  gpsSerial.begin(9600, SERIAL_8N1, RX2_PIN, TX2_PIN);
  Serial.println(F("[GPS] UART2 inicializado (RX=GPIO16, TX=GPIO17)"));

  // Conectar a WiFi
  conectarWiFi();
}

// =========================================================================
//  LOOP PRINCIPAL
// =========================================================================

void loop() {
  // 1. Leer datos del GPS (no bloqueante)
  while (gpsSerial.available() > 0) {
    gps.encode(gpsSerial.read());
  }

  // 2. Actualizar LED segun estado
  actualizarLED();

  // 3. Verificar WiFi
  if (WiFi.status() != WL_CONNECTED) {
    estadoActual = ESPERANDO_WIFI;
    conectarWiFi();
    return;
  }

  // 4. Verificar si tenemos fix GPS
  if (!gps.location.isValid()) {
    estadoActual = ESPERANDO_GPS;
    if (millis() % 10000 == 0) {
      Serial.println(F("[GPS] Esperando fix..."));
      imprimirEstadoGPS();
    }
  } else {
    estadoActual = ENVIANDO_DATOS;
  }

  // 5. Enviar ubicacion al servidor cada INTERVALO_ENVIO_MS
  unsigned long ahora = millis();
  if (ahora - ultimoEnvio >= INTERVALO_ENVIO_MS) {
    ultimoEnvio = ahora;

    if (gps.location.isValid()) {
      enviarUbicacion();
    } else {
      Serial.println(F("[SKIP] Sin fix GPS, no se envia ubicacion"));
    }
  }
}

// =========================================================================
//  CONEXION WIFI
// =========================================================================

void conectarWiFi() {
  estadoActual = ESPERANDO_WIFI;
  Serial.print(F("\n[WiFi] Conectando a "));
  Serial.print(WIFI_SSID);
  Serial.print(F("..."));

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);

  int intentos = 0;
  while (WiFi.status() != WL_CONNECTED && intentos < 40) {
    delay(500);
    Serial.print(F("."));
    intentos++;

    // Seguir leyendo GPS mientras conectamos
    while (gpsSerial.available() > 0) {
      gps.encode(gpsSerial.read());
    }
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println(F("\n[WiFi] Conectado!"));
    Serial.print(F("[WiFi] IP: "));
    Serial.println(WiFi.localIP());
  } else {
    Serial.println(F("\n[WiFi] Fallo al conectar. Reintentando..."));
  }
}

// =========================================================================
//  ENVIO DE UBICACION AL SERVIDOR TECBUS VIA HTTP PUT
// =========================================================================

void enviarUbicacion() {
  HTTPClient http;
  String url = String(SERVER_URL) + "/api/camiones/update-location";

  http.begin(url);
  http.addHeader("Content-Type", "application/json");

  // Construir JSON con los datos
  String jsonData;
  {
    StaticJsonDocument<200> doc;
    doc["busId"] = BUS_ID;
    doc["lat"] = gps.location.lat();
    doc["lng"] = gps.location.lng();
    doc["speed"] = (int)gps.speed.kmph();

    serializeJson(doc, jsonData);
  }

  Serial.print(F("[HTTP] Enviando ubicacion: "));
  Serial.println(jsonData);

  int codigoHTTP = http.PUT(jsonData);

  if (codigoHTTP > 0) {
    Serial.print(F("[HTTP] Respuesta: "));
    Serial.println(codigoHTTP);

    if (codigoHTTP == 200) {
      Serial.println(F("[OK] Ubicacion actualizada correctamente"));
      estadoActual = ENVIANDO_DATOS;
    } else {
      Serial.print(F("[AVISO] Codigo inesperado: "));
      Serial.println(http.getString());
      estadoActual = ERROR_CONEXION;
    }
  } else {
    Serial.print(F("[ERROR] Fallo HTTP: "));
    Serial.println(http.errorToString(codigoHTTP).c_str());
    estadoActual = ERROR_CONEXION;
  }

  http.end();
}

// =========================================================================
//  CONTROL DEL LED DE ESTADO
// =========================================================================

void actualizarLED() {
  unsigned long ahora = millis();
  unsigned long intervalo = 0;

  switch (estadoActual) {
    case ESPERANDO_WIFI:
      intervalo = 100;  // Parpadeo rapido (cada 100ms)
      break;
    case ESPERANDO_GPS:
      intervalo = 500;  // Parpadeo lento (cada 500ms)
      break;
    case ENVIANDO_DATOS:
      // LED fijo encendido
      digitalWrite(LED_BUILTIN, HIGH);
      return;
    case ERROR_CONEXION:
      intervalo = 200;  // Parpadeo medio (cada 200ms)
      break;
  }

  if (ahora - ultimoParpadeo >= intervalo) {
    ultimoParpadeo = ahora;
    ledEstado = !ledEstado;
    digitalWrite(LED_BUILTIN, ledEstado);
  }
}

// =========================================================================
//  INFORMACION DE DEPURACION DEL GPS
// =========================================================================

void imprimirEstadoGPS() {
  Serial.print(F("[GPS] Satelites: "));
  Serial.print(gps.satellites.value());
  Serial.print(F(" | HDOP: "));
  Serial.print(gps.hdop.hdop());
  Serial.print(F(" | Fecha: "));
  if (gps.date.isValid()) {
    Serial.print(gps.date.day());
    Serial.print(F("/"));
    Serial.print(gps.date.month());
    Serial.print(F("/"));
    Serial.print(gps.date.year());
  } else {
    Serial.print(F("--/--/----"));
  }
  Serial.print(F(" | Hora: "));
  if (gps.time.isValid()) {
    if (gps.time.hour() < 10) Serial.print(F("0"));
    Serial.print(gps.time.hour());
    Serial.print(F(":"));
    if (gps.time.minute() < 10) Serial.print(F("0"));
    Serial.print(gps.time.minute());
    Serial.print(F(":"));
    if (gps.time.second() < 10) Serial.print(F("0"));
    Serial.print(gps.time.second());
  } else {
    Serial.print(F("--:--:--"));
  }
  Serial.println();
}

// =========================================================================
//  INFORMACION COMPLETA DEL GPS (cuando hay fix)

  Serial.println(F("\n========== DATOS GPS =========="));
  Serial.print(F("Latitud:   "));

  Serial.print(F("Altitud:   "));

  Serial.print(gps.speed.kmph());
  Serial.println(F(" km/h"));
  Serial.print(F("Direccion: "));
  Serial.println(gps.course.deg());
  Serial.print(F("Satelites: "));
  Serial.println(gps.satellites.value());
  Serial.print(F("HDOP:      "));
  Serial.println(gps.hdop.hdop());
  Serial.print(F("Fecha:     "));
  Serial.print(gps.date.day());
  Serial.print(F("/"));
  Serial.print(gps.date.month());
  Serial.print(F("/"));
  Serial.println(gps.date.year());
  Serial.print(F("Hora:      "));
  Serial.print(gps.time.hour());
  Serial.print(F(":"));
  Serial.print(gps.time.minute());
  Serial.print(F(":"));
  Serial.println(gps.time.second());
  Serial.println(F("================================\n"));
}
